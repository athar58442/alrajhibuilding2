window.onload = function() {
    var img = document.getElementById('img');
    var cvs = document.getElementById('cvs');
    cvs.width = img.naturalWidth;
    cvs.height = img.naturalHeight;
    var ctx = cvs.getContext('2d');
    ctx.drawImage(img, 0, 0);
    var imgData = ctx.getImageData(0, 0, cvs.width, cvs.height);
    var data = imgData.data;

    // Sample corners and center
    var corners = [
        "top-left: " + data[0] + "," + data[1] + "," + data[2] + "," + data[3],
        "center: " + data[(Math.floor(cvs.height/2)*cvs.width + Math.floor(cvs.width/2))*4] + "," +
                     data[(Math.floor(cvs.height/2)*cvs.width + Math.floor(cvs.width/2))*4 + 1] + "," +
                     data[(Math.floor(cvs.height/2)*cvs.width + Math.floor(cvs.width/2))*4 + 2] + "," +
                     data[(Math.floor(cvs.height/2)*cvs.width + Math.floor(cvs.width/2))*4 + 3]
    ];

    // Find non-transparent or non-white color distribution
    var colorSamples = {};
    for (var i = 0; i < data.length; i += 16) {
        var r = data[i], g = data[i+1], b = data[i+2], a = data[i+3];
        if (a > 20) {
            var key = Math.round(r/30)*30 + "," + Math.round(g/30)*30 + "," + Math.round(b/30)*30;
            colorSamples[key] = (colorSamples[key] || 0) + 1;
        }
    }

    var p = document.createElement('p');
    p.id = "info";
    p.innerText = JSON.stringify({ corners: corners, samples: colorSamples });
    document.body.appendChild(p);
};
</script>
